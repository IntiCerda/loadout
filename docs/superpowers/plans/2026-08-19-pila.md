# Pila Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a public web app where a developer picks tools, extensions, fonts, global libraries, and local AI models from a curated catalog and receives one readable PowerShell script — as a `.ps1` download or an `irm | iex` one-liner.

**Architecture:** A typed catalog module is the single source of truth. Two pure functions — `resolve` (expand dependencies, sum size) and `generateScript` (group by installer, emit PowerShell) — are consumed by both the client-side live preview and the `/api/script` route handler, so the previewed script is byte-identical to the delivered one. Selection state lives in the URL query string; there is no database and no session.

**Tech Stack:** Next.js 16 (App Router), TypeScript, Tailwind v4, shadcn/ui on Base UI, Lucide, Vitest, Vercel.

- **Target OS: Windows 10 (build 19041+) and Windows 11.** Windows 10 is the harder target and the one to verify against — `winget` is preinstalled on 11 but requires App Installer on 10, and `wsl --install` needs build 19041+. The generated script must preflight both and print an actionable message rather than failing on line one.
- **A Linux (`apt`) target ships in Phase 5.** The `Item` type must carry Linux targeting from Task 2 onward. Backfilling a field across 80 catalog entries later is the expensive part; the second generator is not.
- Repository root: `C:\Users\PC\Documents\GitHub\pila`. New repo, not a fork of `Script-Dev-W10-W11`.
- All code, comments, identifiers, UI copy, and commit messages in **English**. Conventional commits. Author `Inti` only. Never add `Co-Authored-By` or AI attribution.
- Brand name lives in exactly one constant, `lib/brand.ts`. Working value: `Pila`. Never hardcode the name anywhere else.
- Generated PowerShell must be **ASCII-only** — no accented characters, no box-drawing, no emoji. `irm | iex` decoding is unreliable otherwise.
- Generated PowerShell must contain **no `param()` block** and **no top-level `exit`**. `iex` fails on `param()`, and `exit` closes the user's console.
- The generated script may only contain strings that come from `data/catalog.ts`. Query-string values are used solely as allowlist lookup keys and never interpolated into output.
- No emoji as icons anywhere in the UI. Lucide SVG only.
- Every interactive element: `cursor-pointer`, a visible focus ring, and a 150–300ms transition. Minimum hit target 44×44px.
- All motion must be disabled under `prefers-reduced-motion: reduce`.
- Responsive breakpoints to verify: 375px, 768px, 1024px, 1440px.
- Dark mode is the primary and only theme in v1. Background is `#0F172A`, never `#000000`.

Create `lib/url.ts`:

```ts
/**
 * Hard cap on ids accepted from a query string. Selecting the entire catalog
 * stays well under this; anything above it is a malformed or hostile URL.
 */
export const MAX_IDS = 300

/**
 * Ids must start alphanumeric and may then contain dots, colons, dashes and
 * underscores — enough for Ollama model tags like `qwen2.5-coder:7b`.
 */
const ID_PATTERN = /^[a-z0-9][a-z0-9.\-:_]*$/

export function parseIds(raw: string | null): string[] {
  if (!raw) return []
  const seen = new Set<string>()
  const out: string[] = []
  for (const segment of raw.split(',')) {
    const id = segment.trim().toLowerCase()
    if (!id || seen.has(id) || !ID_PATTERN.test(id)) continue
    seen.add(id)
    out.push(id)
    if (out.length === MAX_IDS) break
  }
  return out
}

export function serializeIds(ids: string[]): string {
  return ids.join(',')
}
```

Create `lib/resolve.ts`:

```ts
import type { Item } from './types'

/**
 * Expand a selection into the full set of items that must be installed,
 * following `requires` transitively. Unknown ids are dropped — the catalog is
 * the allowlist, which is what keeps query-string input out of the generated
 * script. The result is returned in catalog order so the emitted script is
 * deterministic for a given selection regardless of click order.
 */
export function resolve(ids: string[], items: Item[]): Item[] {
  const byId = new Map(items.map((item) => [item.id, item]))
  const selected = new Set<string>()

  const visit = (id: string): void => {
    if (selected.has(id)) return
    const item = byId.get(id)
    if (!item) return
    selected.add(id)
    for (const dep of item.requires ?? []) visit(dep)
  }

  for (const id of ids) visit(id)

  return items.filter((item) => selected.has(item.id))
}

export function totalSizeMb(items: Item[]): number {
  return items.reduce((sum, item) => sum + (item.sizeMb ?? 0), 0)
}

export function formatSize(mb: number): string {
  if (mb <= 0) return '--'
  if (mb < 1024) return `${Math.round(mb)} MB`
  return `${(mb / 1024).toFixed(1)} GB`
}
```

```ts
import type { Installer, Item } from './types'
import { totalSizeMb, formatSize } from './resolve'
import { BRAND } from './brand'

/**
 * Phase order matters. winget runs first because it installs the commands the
 * later phases call (code, npm, ollama). PATH is refreshed straight after it.
 * Ollama model pulls run last because they are by far the largest downloads,
 * so everything else is already usable if the user interrupts.
 */
/**
 * Installers the Windows emitter handles. Excludes the Linux-only variants so
 * that adding one to `Installer` for the Phase 5 bash target can never silently
 * widen these Records and demand a PowerShell emitter that must not exist.
 * `'script'` is pre-excluded; it joins `Installer` in Task 14.
 */
type WindowsInstaller = Exclude<Installer, 'apt' | 'script'>

const PHASES: { installer: WindowsInstaller; label: string }[] = [
  { installer: 'winget', label: 'Installing applications' },
  { installer: 'wsl', label: 'Installing WSL distros' },
  { installer: 'font', label: 'Installing fonts' },
  { installer: 'vscode', label: 'Installing VS Code extensions' },
  { installer: 'npm', label: 'Installing global npm packages' },
  { installer: 'pipx', label: 'Installing pipx packages' },
  { installer: 'ollama', label: 'Pulling Ollama models' },
  { installer: 'claude-plugin', label: 'Installing Claude Code plugins' },
]

const HELPERS: Record<WindowsInstaller, string> = {
  winget: `function Install-WingetPackage([string]$Id) {
    $found = winget list --id $Id --exact --accept-source-agreements 2>$null |
             Select-String ([regex]::Escape($Id))
    if ($found) {
        Write-Host "  $Id already installed, skipping." -ForegroundColor DarkGray
        return

        --accept-source-agreements --accept-package-agreements

 * commands that winget installed moments earlier in the same run.

            [Environment]::GetEnvironmentVariable('Path','User')`

 * Windows 10 preflight. winget ships with Windows 11 but on Windows 10 it
 * arrives via App Installer from the Store, so a clean Win10 box fails on the
 * very first winget call with an unhelpful error. Emitted only when the
 * selection actually contains winget items.
 */

    Write-Host "winget is not available on this machine." -ForegroundColor Red
    Write-Host "Windows 11 ships it. On Windows 10, install 'App Installer' from the" -ForegroundColor Red

    Write-Host "  ms-windows-store://pdp/?ProductId=9NBLGGH4NNS1" -ForegroundColor Yellow
    return
}`

/**
 * WSL preflight. `wsl --install` needs Windows 10 build 19041 or newer.
 * Emitted only when the selection contains wsl items.
 */
const WSL_PREFLIGHT = `# Initialise before testing. Under \`irm | iex\` this runs in the CALLER'S
# session scope, so a second run in the same console would otherwise inherit a
# stale $true from the first and skip WSL forever. Also required under
# Set-StrictMode -Version Latest, where reading an unset variable throws.
$script:SkipWsl = $false
$build = [int](Get-CimInstance Win32_OperatingSystem).BuildNumber
if ($build -lt 19041) {
    Write-Host "WSL2 needs Windows 10 build 19041 or newer (found $build)." -ForegroundColor Yellow
    Write-Host "Skipping the Linux distro steps. Windows Update will get you there." -ForegroundColor Yellow
    $script:SkipWsl = $true
}`

const TRANSCRIPT = `$logFile = Join-Path $env:TEMP ('setup-' + (Get-Date -Format 'yyyyMMdd-HHmmss') + '.log')
try { Start-Transcript -Path $logFile -Append | Out-Null } catch { }
Write-Host "Log: $logFile" -ForegroundColor DarkGray`

/**
 * Emit a self-contained PowerShell script for the given resolved items.
 *
 * Contract, enforced by lib/generate.test.ts:
 *  - no `param()` block: `iex` rejects scripts that declare parameters
 *  - no top-level `exit`: it would close the user's console under `iex`
 *  - ASCII only: `irm | iex` decoding of non-ASCII is unreliable
 *  - every emitted string comes from the catalog, never from user input
 */
export function generateScript(items: Item[], shareUrl: string): string {
  const sizeLabel = formatSize(totalSizeMb(items))
  const header = [
    `# ${BRAND} -- generated setup script`,
    `# ${shareUrl}`,
    `# ${items.length} item(s), about ${sizeLabel} to download`,
    `#`,
    `# This is plain PowerShell. Read it before you run it -- nothing is hidden.`,
    '',
  ].join('\n')

  if (items.length === 0) {
    return `${header}Write-Host "Nothing selected. Pick some tools first." -ForegroundColor Yellow\n`
  }

  const active = PHASES.map((phase) => ({
    ...phase,
    items: items.filter((item) => item.installer === phase.installer),
  })).filter((phase) => phase.items.length > 0)

  const usedInstallers = active.map((phase) => phase.installer)
  const helpers = usedInstallers.map((installer) => HELPERS[installer]).join('\n\n')

  // Preflights are per-installer: a models-only run has no reason to fail on a
  // Windows 10 App Installer check it never needed.
  const preflights = [
    usedInstallers.includes('winget') ? WINGET_PREFLIGHT : '',
    usedInstallers.includes('wsl') ? WSL_PREFLIGHT : '',
  ]
    .filter(Boolean)
    .join('\n\n')

  const blocks = active.map((phase, index) => {
    const step = `Write-Host "[${index + 1}/${active.length}] ${phase.label}..." -ForegroundColor Cyan`
    const calls = phase.items
      .map((item) => `${CALLS[phase.installer](item)}  # ${item.name}`)
      .join('\n')
    const refresh = phase.installer === 'winget' ? `\n\n${REFRESH_PATH}` : ''
    return `${step}\n${calls}${refresh}`
  })

  return [
    header,
    ADMIN_GATE,
    '',
    preflights,
    '',
    TRANSCRIPT,
    '',
    helpers,
    '',
    ...blocks.map((block) => `${block}\n`),
    `Write-Host ""`,
    `Write-Host "Done. Restart your terminal so PATH changes take effect." -ForegroundColor Green`,
    `try { Stop-Transcript | Out-Null } catch { }`,
    '',
  ].join('\n')
}
```

```ts
import { catalog } from '@/data/catalog'
import { BRAND, SITE_URL } from '@/lib/brand'
import { generateScript } from '@/lib/generate'
import { resolve } from '@/lib/resolve'
import { parseIds, serializeIds } from '@/lib/url'

const FILENAME = `${BRAND.toLowerCase()}-setup.ps1`

export function GET(req: Request): Response {
  const url = new URL(req.url)

  // Merge repeated `p` parameters rather than letting the first one win.
  // `?p=&p=git` would otherwise serve an empty script because the empty first
  // value shadows the real one.
  const ids = parseIds(url.searchParams.getAll('p').join(','))
  const items = resolve(ids, catalog)

  // Two independent reasons this is derived from the request rather than from
  // SITE_URL. First, NEXT_PUBLIC_SITE_URL is inlined into the client bundle at
  // BUILD time but read here at RUNTIME, so setting it after a deploy makes the
  // server emit one host and the live preview emit another — the previewed and
  // delivered scripts stop being byte-identical, and every local check still
  // passes. Second, on a Vercel preview deployment SITE_URL points at
  // production, so a one-liner copied from a preview would install from prod.
  const origin = url.origin || SITE_URL

  // Built from resolved ids, not from the raw query — the script always links
  // back to a selection that actually exists in the catalog. This is what keeps
  // query-string input out of the generated script entirely.
  const shareUrl = `${origin}/?p=${serializeIds(items.map((item) => item.id))}`
  const script = generateScript(items, shareUrl)

  const headers = new Headers({
    'Content-Type': 'text/plain; charset=utf-8',
    'Cache-Control': 'public, max-age=3600, s-maxage=86400',
  })

  if (url.searchParams.get('download') === '1') {
    headers.set('Content-Disposition', `attachment; filename="${FILENAME}"`)
  }

  return new Response(script, { headers })
}
```

|---|---|

```tsx
'use client'

import { useCallback, useEffect, useMemo, useState } from 'react'
import { catalog } from '@/data/catalog'
import { generateScript } from '@/lib/generate'
import { resolve, totalSizeMb } from '@/lib/resolve'
import { parseIds, serializeIds } from '@/lib/url'
import { SITE_URL } from '@/lib/brand'
import { CatalogGrid } from '@/components/catalog-grid'

export default function Page() {
  const [selectedIds, setSelectedIds] = useState<string[]>([])

  // Read the initial selection from the URL once, on mount. Reading it during
  // render would desync server and client HTML.
  useEffect(() => {
    const params = new URLSearchParams(window.location.search)
    setSelectedIds(parseIds(params.get('p')))
  }, [])

  // Mirror selection into the URL without a navigation, so the address bar is
  // always a shareable link.
  useEffect(() => {
    const query = serializeIds(selectedIds)
    const next = query ? `${window.location.pathname}?p=${query}` : window.location.pathname
    window.history.replaceState(null, '', next)
  }, [selectedIds])

  const toggle = useCallback((id: string) => {
    setSelectedIds((prev) =>
      prev.includes(id) ? prev.filter((existing) => existing !== id) : [...prev, id],
    )
  }, [])

  const resolved = useMemo(() => resolve(selectedIds, catalog), [selectedIds])
  const selectedSet = useMemo(() => new Set(selectedIds), [selectedIds])
  const requiredSet = useMemo(
    () => new Set(resolved.map((item) => item.id).filter((id) => !selectedSet.has(id))),
    [resolved, selectedSet],
  )

  // Must match how app/api/script/route.ts builds it, or the previewed script
  // stops being byte-identical to the delivered one. Both derive from the live
  // origin; SITE_URL is build-time-inlined here and would skew after an env
  // change or on a preview deployment.
  const [origin, setOrigin] = useState(SITE_URL)
  useEffect(() => setOrigin(window.location.origin), [])

  const shareUrl = `${origin}/?p=${serializeIds(resolved.map((item) => item.id))}`
  const script = useMemo(() => generateScript(resolved, shareUrl), [resolved, shareUrl])
  const sizeMb = useMemo(() => totalSizeMb(resolved), [resolved])

  return (
    <main className="mx-auto w-full max-w-7xl px-4 py-12 sm:px-6 lg:px-8">
      <CatalogGrid
        items={catalog}
        selectedIds={selectedSet}
        requiredIds={requiredSet}
        onToggle={toggle}
      />
      {/* Kit sidebar and script preview land in Tasks 8 and 9. */}
      <pre className="sr-only">{script}</pre>
      <span className="sr-only">{sizeMb}</span>
    </main>
  )
}
```

- [ ] **Step 1: Build the preview**

Syntax colouring is done with three regular expressions rather than a
highlighter dependency. The generated PowerShell is emitted by code we own, so
the token set is small, known and closed — a full highlighter would be roughly
40× the bundle for the same four colours.

Create `components/script-preview.tsx`:

```tsx
'use client'

import { useState } from 'react'
import { ChevronDown } from 'lucide-react'

const COMMENT = /(#.*$)/
const STRING = /("[^"]*")/
const KEYWORD = /\b(function|if|else|return|foreach|try|catch|not|param)\b/

function colourise(line: string): React.ReactNode {
  if (COMMENT.test(line) && line.trim().startsWith('#')) {
    return <span className="text-foreground/35">{line}</span>
  }
  const parts = line.split(new RegExp(`${STRING.source}|${KEYWORD.source}`, 'g'))
  return parts.map((part, index) => {
    if (!part) return null
    if (STRING.test(part)) return <span key={index} className="text-accent">{part}</span>
    if (KEYWORD.test(part)) return <span key={index} className="text-sky-400">{part}</span>
    return <span key={index}>{part}</span>
  })
}

export function ScriptPreview({ script }: { script: string }) {
  const [open, setOpen] = useState(false)
  const lines = script.split('\n')

  return (
    <div className="mt-5 border-t border-border pt-4">
      <button
        type="button"
        onClick={() => setOpen((value) => !value)}
        aria-expanded={open}
        aria-controls="script-body"
        className="flex min-h-[44px] w-full cursor-pointer items-center justify-between gap-2
          rounded-lg px-1 text-left text-sm font-medium transition-colors duration-200
          hover:text-accent focus-visible:ring-2 focus-visible:ring-accent
          focus-visible:ring-offset-2 focus-visible:ring-offset-primary"
      >
        <span>View script</span>
        <span className="flex items-center gap-2 font-mono text-xs text-foreground/40">
          {lines.length} lines
          <ChevronDown
            aria-hidden
            className={`size-4 transition-transform duration-200 ${open ? 'rotate-180' : ''}`}
          />
        </span>
      </button>

      {open ? (
        <pre
          id="script-body"
          tabIndex={0}
          className="mt-2 max-h-[60vh] overflow-auto rounded-lg border border-border
            bg-background p-4 font-mono text-xs leading-relaxed
            focus-visible:ring-2 focus-visible:ring-accent"
        >
          <code>
            {lines.map((line, index) => (
              <div key={index} className="whitespace-pre">
                {colourise(line)}
              </div>
            ))}
          </code>
        </pre>
      ) : null}
    </div>
  )
}
```

## Phase 5 — The Linux target

Ships after Windows is live. Windows is the harder platform to get right and
the one the catalog was authored against; shipping it first means Linux is
additive rather than a simultaneous two-front build.

**What is free and what is not.** Five installers — `vscode`, `npm`, `pipx`,
`ollama`, `claude-plugin` — are byte-identical on Linux. The generator for them

system-package layer: several desktop-grade packages are not in Debian's
default repositories at all (VS Code, Docker, Ollama each publish their own
repo or install script). That is why `linux.installer` has two values, and why
Task 15 is separate from Task 14.

- [ ] **Step 4: Implement the bash generator**

Create `lib/generate-linux.ts`:

```ts
import type { Installer, Item } from './types'
import { PORTABLE_INSTALLERS } from './types'
import { totalSizeMb, formatSize } from './resolve'
import { BRAND } from './brand'

/**
 * An item runs on Linux when its installer is portable, or when it is a
 * `winget` item that declares a Linux equivalent. `wsl` items never do.
 */
export function linuxSupported(item: Item): boolean {
  if (item.installer === 'wsl') return false
  if (PORTABLE_INSTALLERS.includes(item.installer)) return true
  return item.linux !== undefined
}

/**
 * `curl | bash` gives the script no controlling terminal. Anything that reads
 * stdin — an apt confirmation prompt, a sudo password prompt — hangs forever
 * with no output. This is the Linux mirror of the Windows `param()` trap:
 * every command below must be non-interactive by construction.
 */
const ROOT_GATE = `if [ "$(id -u)" -ne 0 ]; then
  echo "This script installs system packages and must run as root." >&2
  echo "Re-run it like this:" >&2
  echo "  sudo bash -c \\"\\$(curl -fsSL <url>)\\"" >&2
  exit 1
fi`

const HELPERS: Partial<Record<Installer | 'script', string>> = {
  apt: `apt_install() {
  if dpkg -s "$1" >/dev/null 2>&1; then
    echo "  $1 already installed, skipping."
    return
  fi
  DEBIAN_FRONTEND=noninteractive apt-get install -y --no-install-recommends "$1"
}`,

  script: `script_install() {
  echo "  running vendor install script: $1"
  curl -fsSL "$1" | DEBIAN_FRONTEND=noninteractive bash
}`,

  vscode: `vscode_ext() {
  if ! command -v code >/dev/null 2>&1; then
    echo "  code CLI not found, skipping $1."
    return
  fi
  code --install-extension "$1" --force --no-sandbox --user-data-dir /tmp/pila-code
}`,

  npm: `npm_global() {
  if ! command -v npm >/dev/null 2>&1; then
    echo "  npm not found, skipping $1."
    return
  fi
  npm install -g "$1"
}`,

  pipx: `pipx_install() {
  if ! command -v pipx >/dev/null 2>&1; then
    DEBIAN_FRONTEND=noninteractive apt-get install -y pipx
  fi
  pipx install "$1"
}`,

  ollama: `ollama_pull() {
  if ! command -v ollama >/dev/null 2>&1; then
    echo "  ollama not found, skipping $1."
    return
  fi
  if ollama list 2>/dev/null | grep -qF "$1"; then
    echo "  $1 already pulled, skipping."
    return
  fi
  ollama pull "$1"
}`,

  'claude-plugin': `claude_plugin() {
  if ! command -v claude >/dev/null 2>&1; then
    echo "  claude CLI not found, skipping $1."
    return
  fi
  claude plugin install "$1"
}`,

  font: `font_install() {
  dir="$HOME/.local/share/fonts"
  mkdir -p "$dir"
  tmp="$(mktemp -d)"
  curl -fsSL "$1" -o "$tmp/font.zip"
  unzip -oq "$tmp/font.zip" -d "$tmp"
  find "$tmp" \\( -name '*.ttf' -o -name '*.otf' \\) -exec cp -n {} "$dir/" \\;
  fc-cache -f >/dev/null
  rm -rf "$tmp"
}`,
}

/** Same ordering rationale as the Windows generator: system packages first so
 *  later phases find the commands they need, model pulls last because they are
 *  the largest downloads. */
const PHASES: { key: string; label: string; call: (item: Item) => string }[] = [
  { key: 'apt', label: 'Installing system packages', call: (i) => `apt_install "${i.linux!.ref}"` },
  { key: 'script', label: 'Running vendor installers', call: (i) => `script_install "${i.linux!.ref}"` },
  { key: 'font', label: 'Installing fonts', call: (i) => `font_install "${i.ref}"` },
  { key: 'vscode', label: 'Installing VS Code extensions', call: (i) => `vscode_ext "${i.ref}"` },
  { key: 'npm', label: 'Installing global npm packages', call: (i) => `npm_global "${i.ref}"` },
  { key: 'pipx', label: 'Installing pipx packages', call: (i) => `pipx_install "${i.ref}"` },
  { key: 'ollama', label: 'Pulling Ollama models', call: (i) => `ollama_pull "${i.ref}"` },
  { key: 'claude-plugin', label: 'Installing Claude Code plugins', call: (i) => `claude_plugin "${i.ref}"` },
]

/** Which phase an item belongs to: winget items route by their linux
 *  installer, everything else by its own. */
function phaseKey(item: Item): string {
  return item.installer === 'winget' ? item.linux!.installer : item.installer
}

export function generateBash(items: Item[], shareUrl: string): string {
  const supported = items.filter(linuxSupported)
  const dropped = items.filter((item) => !linuxSupported(item))

  const header = [
    '#!/usr/bin/env bash',
    `# ${BRAND} -- generated setup script (Debian/Ubuntu)`,
    `# ${shareUrl}`,
    `# ${supported.length} item(s), about ${formatSize(totalSizeMb(supported))} to download`,
    '#',
    '# This is plain bash. Read it before you run it -- nothing is hidden.',
    '',
  ].join('\n')

  const notes = dropped.length
    ? dropped.map((item) => `echo "  skipped: ${item.name} -- not available on Linux"`).join('\n') + '\n'
    : ''

  if (supported.length === 0) {
    return `${header}echo "Nothing to install for this target."\n${notes}`
  }

  const active = PHASES.map((phase) => ({
    ...phase,
    items: supported.filter((item) => phaseKey(item) === phase.key),
  })).filter((phase) => phase.items.length > 0)

  const helpers = active.map((phase) => HELPERS[phase.key as Installer]).join('\n\n')
  const needsApt = active.some((phase) => phase.key === 'apt' || phase.key === 'pipx')

  const blocks = active.map((phase, index) => {
    const step = `echo "[${index + 1}/${active.length}] ${phase.label}..."`
    const calls = phase.items.map((item) => `${phase.call(item)}  # ${item.name}`).join('\n')
    return `${step}\n${calls}\n`
  })

  return [
    header,
    ROOT_GATE,
    '',
    'export DEBIAN_FRONTEND=noninteractive',
    needsApt ? 'apt-get update -y\n' : '',
    helpers,
    '',
    notes,
    ...blocks,
    'echo ""',
    'echo "Done. Open a new shell so PATH changes take effect."',
    '',
  ]
    .filter((part) => part !== '')
    .join('\n')
}
```
